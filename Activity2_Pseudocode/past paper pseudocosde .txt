INPUT playerID

INPUT numGames
WHILE numGames < 5 OR numGames > 10
    OUTPUT "Invalid. Enter 5 to 10 games"
    INPUT numGames
ENDWHILE

FOR i = 0 TO numGames - 1
    INPUT scores[i]
    INPUT times[i]
    WHILE times[i] < 5 OR times[i] > 60
        OUTPUT "Invalid. Enter 5 to 60 minutes"
        INPUT times[i]
    ENDWHILE
NEXT i

highestScore = scores[0]
totalTime = 0
FOR i = 0 TO numGames - 1
    IF scores[i] > highestScore THEN
        highestScore = scores[i]
    ENDIF
    totalTime = totalTime + times[i]
NEXT i

averageTime = ROUND(totalTime / numGames)

OUTPUT playerID
OUTPUT highestScore
OUTPUT averageTime

OPEN FILE "playerdata.txt"
WRITE playerID, scores[], times[], highestScore, averageTime
CLOSE FILE

END
